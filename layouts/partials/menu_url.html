{{- $url := .Entry.URL -}} {{- if .Entry.Page -}} {{- $url = .Entry.Page.RelPermalink -}} {{- else if and (not (urls.Parse $url).Scheme) (not (hasPrefix $url "//")) (not (hasPrefix $url "#")) -}} {{- $url = .Entry.MenuConfig.URL -}} {{- $base := strings.TrimSuffix "/" (urls.Parse .Page.Site.BaseURL).Path -}} {{- if and $base (hasPrefix $url (printf "%s/" $base)) -}} {{- $url = strings.TrimPrefix $base $url -}} {{- end -}} {{- $home := strings.TrimPrefix $base .Page.Site.Home.RelPermalink -}} {{- if or (eq $url (strings.TrimSuffix "/" $home)) (hasPrefix $url $home) -}} {{- $url = strings.TrimPrefix "/" $url | relURL -}} {{- else -}} {{- $url = strings.TrimPrefix "/" $url | relLangURL -}} {{- end -}} {{- end -}} {{- return $url -}}
