{{- $shortcode := .Shortcode -}} {{- $inner := .Inner -}} {{- with $shortcode -}} {{- $direction := .Get "dir" | default "auto" -}} {{- if not (in (slice "ltr" "rtl" "auto") $direction) -}} {{- errorf "%s shortcode: invalid direction %q at %s" .Name $direction .Position -}} {{- end -}} {{- $tag := "div" -}} {{- $display := "block" -}} {{- if eq .Name "bidi" -}} {{- $tag = "bdi" -}} {{- $display = "inline" -}} {{- end -}} {{- $content := .Page.RenderString (dict "display" $display) $inner -}} {{- range $token, $fragment := .Store.Get "directionFragments" -}} {{- $content = replace $content $token $fragment -}} {{- end -}} {{- $language := "" -}} {{- with .Get "lang" -}} {{- $language = printf ` lang="%s"` (htmlEscape .) -}} {{- end -}} {{- $result := printf `<%s dir="%s"%s>%s` $tag $direction $language $content $tag | safeHTML -}} {{- with .Parent -}} {{- if in (slice "bidi" "direction") .Name -}} {{- $token := printf "OCTODIRECTION%dTOKEN" $shortcode.Ordinal -}} {{- .Store.SetInMap "directionFragments" $token $result -}} {{- $result = $token -}} {{- end -}} {{- end -}} {{- return $result -}} {{- end -}}
